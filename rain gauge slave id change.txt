#include <Arduino.h>
#include <HardwareSerial.h>

#define RXD2 14
#define TXD2 13

HardwareSerial modbusSerial(2);

const uint8_t OLD_ID = 0x01;
const uint8_t NEW_ID = 0x04;

const uint16_t ID_REGISTER = 0x0030;

// --------------------------------------------------
// Modbus CRC16
// --------------------------------------------------
uint16_t crc16(const uint8_t *data, uint16_t len)
{
    uint16_t crc = 0xFFFF;

    for (uint16_t i = 0; i < len; i++)
    {
        crc ^= data[i];

        for (uint8_t j = 0; j < 8; j++)
        {
            if (crc & 1)
            {
                crc >>= 1;
                crc ^= 0xA001;
            }
            else
            {
                crc >>= 1;
            }
        }
    }

    return crc;
}

// --------------------------------------------------
// Change ID
//
// 01 06 00 30 00 04 CRC CRC
//
// Function 06 = Write Single Register
// Register    = 0x0030
// New ID      = 4
// --------------------------------------------------
void changeSlaveID()
{
    uint8_t cmd[8];

    cmd[0] = OLD_ID;
    cmd[1] = 0x06;

    cmd[2] = 0x00;
    cmd[3] = 0x30;

    cmd[4] = 0x00;
    cmd[5] = NEW_ID;

    uint16_t crc = crc16(cmd, 6);

    cmd[6] = crc & 0xFF;
    cmd[7] = crc >> 8;

    Serial.println("Changing Slave ID 1 -> 4");

    Serial.print("TX: ");

    for (int i = 0; i < 8; i++)
    {
        Serial.printf("%02X ", cmd[i]);
    }

    Serial.println();

    while (modbusSerial.available())
        modbusSerial.read();

    modbusSerial.write(cmd, 8);
    modbusSerial.flush();

    // Wait for response
    uint8_t response[16];
    uint8_t len = 0;

    unsigned long start = millis();

    while (millis() - start < 500)
    {
        while (modbusSerial.available())
        {
            if (len < sizeof(response))
                response[len++] = modbusSerial.read();
            else
                modbusSerial.read();
        }

        if (len >= 8)
            break;
    }

    if (len < 8)
    {
        Serial.println("No response.");
        return;
    }

    Serial.print("RX: ");

    for (int i = 0; i < len; i++)
    {
        Serial.printf("%02X ", response[i]);
    }

    Serial.println();

    // Check CRC
    uint16_t receivedCRC =
        response[len - 2] |
        ((uint16_t)response[len - 1] << 8);

    uint16_t calculatedCRC =
        crc16(response, len - 2);

    if (receivedCRC != calculatedCRC)
    {
        Serial.println("CRC ERROR");
        return;
    }

    // Check normal response
    if (response[0] == OLD_ID &&
        response[1] == 0x06 &&
        response[2] == 0x00 &&
        response[3] == 0x30 &&
        response[4] == 0x00 &&
        response[5] == NEW_ID)
    {
        Serial.println();
        Serial.println("SUCCESS!");
        Serial.println("Slave ID changed: 1 -> 4");
        Serial.println();
        Serial.println("Power-cycle the sensor before testing.");
    }
    else
    {
        Serial.println("Unexpected response.");
    }
}

// --------------------------------------------------
// Setup
// --------------------------------------------------
void setup()
{
    Serial.begin(115200);

    modbusSerial.begin(
        9600,
        SERIAL_8N1,
        RXD2,
        TXD2
    );

    delay(1000);

    Serial.println();
    Serial.println("WS-601 SH Slave ID Changer");
    Serial.println("---------------------------");

    changeSlaveID();
}

void loop()
{
    // One-time operation
}
