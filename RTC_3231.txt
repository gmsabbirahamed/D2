#include <Wire.h>
#include <RTClib.h>

RTC_DS3231 rtc;

void setup() {
  Serial.begin(115200);

  Wire.begin(21, 22);

  if (!rtc.begin()) {
    Serial.println("Couldn't find DS3231!");
    while (1);
  }

  // Set the RTC time to the time this sketch was compiled.
  // Uncomment this line ONCE when initially setting the RTC.
  // rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
  // rtc.adjust(DateTime(2026, 10, 6, 15, 17, 0));

}

void loop() {
  DateTime now = rtc.now();

  Serial.print(now.year());
  Serial.print("-");
  Serial.print(now.month());
  Serial.print("-");
  Serial.print(now.day());

  Serial.print(" ");

  Serial.print(now.hour());
  Serial.print(":");
  Serial.print(now.minute());
  Serial.print(":");
  Serial.println(now.second());

  delay(1000);
}
